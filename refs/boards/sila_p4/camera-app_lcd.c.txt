#include "esp_ldo_regulator.h"
#include "esp_lcd_mipi_dsi.h"
#include "esp_lcd_panel_ops.h"
#include "esp_log.h"
#include "esp_system.h"
#include "esp_lcd_st7703.h"
#include "app_lcd.h"
#include "driver/gpio.h"
#include "driver/ledc.h"

static const char *TAG = "app_lcd";

#define EXAMPLE_LDO_MIPI_CHAN               (3)
#define EXAMPLE_LDO_MIPI_VOLTAGE_MV         (2500)

#define EXAMPLE_PIN_NUM_LCD_RST             GPIO_NUM_24

// PWM背光控制配置
#define LEDC_TIMER                          LEDC_TIMER_0
#define LEDC_MODE                           LEDC_LOW_SPEED_MODE
#define DISPLAY_BACKLIGHT_PIN               GPIO_NUM_29    // 背光PWM输出GPIO
#define LEDC_CHANNEL                        LEDC_CHANNEL_0
#define LEDC_DUTY_RES                       LEDC_TIMER_13_BIT
#define LEDC_FREQUENCY                      (5000)
#define LEDC_DUTY_MAX                       (8191)  // 100% (2^13-1)
#define LEDC_DUTY_MIN                       (0)     // 0%

// 背光控制电平定义
#define EXAMPLE_LCD_BK_LIGHT_ON_LEVEL       1
#define EXAMPLE_LCD_BK_LIGHT_OFF_LEVEL      0

static esp_lcd_dsi_bus_handle_t mipi_dsi_bus;
static esp_lcd_panel_io_handle_t mipi_dbi_io;
static esp_lcd_panel_handle_t display_handle;

static void lcd_ldo_power_on(void)
{
    esp_ldo_channel_handle_t ldo_mipi_phy = NULL;
    esp_ldo_channel_config_t ldo_mipi_phy_config = {
        .chan_id = EXAMPLE_LDO_MIPI_CHAN,
        .voltage_mv = EXAMPLE_LDO_MIPI_VOLTAGE_MV,
    };
    ESP_ERROR_CHECK(esp_ldo_acquire_channel(&ldo_mipi_phy_config, &ldo_mipi_phy));
}

// 初始化背光PWM控制
static esp_err_t init_lcd_backlight(void)
{
    // 配置LEDC定时器
    ledc_timer_config_t ledc_timer = {
        .speed_mode       = LEDC_MODE,
        .timer_num        = LEDC_TIMER,
        .duty_resolution  = LEDC_DUTY_RES,
        .freq_hz         = LEDC_FREQUENCY,
        .clk_cfg         = LEDC_AUTO_CLK
    };
    esp_err_t ret = ledc_timer_config(&ledc_timer);
    if (ret != ESP_OK) {
        return ret;
    }

    // 配置LEDC通道
    ledc_channel_config_t ledc_channel = {
        .speed_mode     = LEDC_MODE,
        .channel        = LEDC_CHANNEL,
        .timer_sel      = LEDC_TIMER,
        .intr_type      = LEDC_INTR_DISABLE,
        .gpio_num       = DISPLAY_BACKLIGHT_PIN,
        .duty           = 0,  // 初始为关闭
        .hpoint        = 0
    };
    return ledc_channel_config(&ledc_channel);
}

// 设置背光亮度
static esp_err_t set_lcd_backlight(uint32_t level)
{
    uint32_t duty = level ? LEDC_DUTY_MAX : LEDC_DUTY_MIN;
    esp_err_t ret = ledc_set_duty(LEDC_MODE, LEDC_CHANNEL, duty);
    if (ret == ESP_OK) {
        ret = ledc_update_duty(LEDC_MODE, LEDC_CHANNEL);
    }
    return ret;
}

esp_err_t app_lcd_init(esp_lcd_panel_handle_t *panel_handle)
{
    // Initialize backlight control (PWM)
    ESP_LOGI(TAG, "Initialize LCD backlight control (PWM)");
    ESP_ERROR_CHECK(init_lcd_backlight());
    // 初始化时先关闭背光
    ESP_ERROR_CHECK(set_lcd_backlight(EXAMPLE_LCD_BK_LIGHT_OFF_LEVEL));

    lcd_ldo_power_on();

    ESP_LOGD(TAG, "Install LCD driver");
    esp_lcd_dsi_bus_config_t bus_config = ST7703_PANEL_BUS_DSI_2CH_CONFIG();
    ESP_ERROR_CHECK(esp_lcd_new_dsi_bus(&bus_config, &mipi_dsi_bus));

    ESP_LOGI(TAG, "Install panel IO");
    esp_lcd_dbi_io_config_t dbi_config = ST7703_PANEL_IO_DBI_CONFIG();
    ESP_ERROR_CHECK(esp_lcd_new_panel_io_dbi(mipi_dsi_bus, &dbi_config, &mipi_dbi_io));

    ESP_LOGI(TAG, "Install LCD driver of st7703");
    esp_lcd_dpi_panel_config_t dpi_config = ST7703_720_720_PANEL_60HZ_DPI_CONFIG(EXAMPLE_MIPI_DPI_PX_FORMAT);
    dpi_config.num_fbs = EXAMPLE_LCD_BUF_NUM;
    st7703_vendor_config_t vendor_config = {
        .mipi_config = {
            .dsi_bus = mipi_dsi_bus,
            .dpi_config = &dpi_config,
        },
    };
    const esp_lcd_panel_dev_config_t panel_config = {
        .reset_gpio_num = EXAMPLE_PIN_NUM_LCD_RST,
        .rgb_ele_order = LCD_RGB_ELEMENT_ORDER_RGB,
        .bits_per_pixel = EXAMPLE_LCD_BIT_PER_PIXEL,
        .vendor_config = &vendor_config,
    };
    ESP_ERROR_CHECK(esp_lcd_new_panel_st7703(mipi_dbi_io, &panel_config, &display_handle));
    ESP_ERROR_CHECK(esp_lcd_panel_reset(display_handle));
    ESP_ERROR_CHECK(esp_lcd_panel_init(display_handle));

    *panel_handle = display_handle;

    return ESP_OK;
}

esp_err_t app_lcd_backlight_on(void)
{
    ESP_LOGI(TAG, "Turning on LCD backlight");
    return set_lcd_backlight(EXAMPLE_LCD_BK_LIGHT_ON_LEVEL);
}

esp_err_t app_lcd_backlight_off(void)
{
    ESP_LOGI(TAG, "Turning off LCD backlight");
    return set_lcd_backlight(EXAMPLE_LCD_BK_LIGHT_OFF_LEVEL);
}
