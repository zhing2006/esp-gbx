#include "myes8389.h"

const char* es8389_tag = "es8389";

/* 跨函数共享的设备句柄, 其余接口对象仅在初始化时使用 */
static esp_codec_dev_handle_t s_codec_dev = NULL;

/**
 * @brief       以指定采样率/位宽打开 codec 设备
 * @param       sample_rate : 采样率(Hz)
 * @param       bits        : 位宽(16/24)
 * @retval      0: 成功; 1: 失败
 */
static uint8_t es8389_dev_open(uint32_t sample_rate, uint16_t bits)
{
    esp_codec_dev_sample_info_t sample_cfg = {
        .bits_per_sample = bits,                /* 单声道采样位宽: 16/24 bit */
        .channel         = 2,                   /* 声道数: 立体声 */
        .channel_mask    = 0x03,                /* 声道掩码: bit0/bit1 对应左右声道 */
        .sample_rate     = sample_rate,         /* 采样率(Hz) */
        .mclk_multiple   = 256,                 /* MCLK = 采样率 × 256 */
    };

    if (esp_codec_dev_open(s_codec_dev, &sample_cfg) != ESP_CODEC_DEV_OK)
    {
        ESP_LOGE(es8389_tag, "Open codec device failed(%lu Hz, %u bit)", (unsigned long)sample_rate, bits);
        return 1;
    }

    return 0;
}

/**
 * @brief       ES8389 初始化(I2C 400kHz → codec → esp_codec_dev → 开功放)
 * @param       无
 * @retval      0: 成功; 1: 失败
 */
uint8_t myes8389_init(void)
{
    i2c_master_bus_handle_t i2c_bus_handle = NULL;
    i2c_master_bus_config_t i2c_mst_cfg = {
        .i2c_port   = ES8389_IIC_NUM_PORT,               /* I2C 端口号 */
        .sda_io_num = ES8389_IIC_SDA_GPIO_PIN,           /* SDA 引脚 */
        .scl_io_num = ES8389_IIC_SCL_GPIO_PIN,           /* SCL 引脚 */
        .clk_source = I2C_CLK_SRC_DEFAULT,               /* I2C 时钟源 */
        .glitch_ignore_cnt = 7,                          /* 毛刺滤波宽度(时钟周期) */
        .flags.enable_internal_pullup = true,            /* 启用内部上拉 */
    };
    ESP_ERROR_CHECK(i2c_new_master_bus(&i2c_mst_cfg, &i2c_bus_handle));

    audio_codec_i2c_cfg_t i2c_cfg = {
        .addr           = ES8389_CODEC_DEFAULT_ADDR,     /* codec I2C 地址(8 位形式) */
        .bus_handle     = i2c_bus_handle,                /* I2C 主机总线句柄 */
        .clock_speed_hz = 400 * 1000,                    /* I2C 速率 400kHz */
    };
    const audio_codec_ctrl_if_t *ctrl_if = audio_codec_new_i2c_ctrl(&i2c_cfg);
    assert(ctrl_if);

    audio_codec_i2s_cfg_t i2s_cfg = {
        .port      = 0,                                  /* I2S 端口号(须已由 myi2s 初始化) */
        .rx_handle = NULL,                               /* 接收句柄: 仅播放, 不使用 */
        .tx_handle = tx_handle,                          /* 发送通道句柄(由 myi2s 创建) */
        .clk_src   = 0,                                  /* 时钟源: 0 表示使用默认值 */
    };
    const audio_codec_data_if_t *data_if = audio_codec_new_i2s_data(&i2s_cfg);
    assert(data_if);

    const audio_codec_gpio_if_t *gpio_if = audio_codec_new_gpio();
    assert(gpio_if);
    es8389_codec_cfg_t es8389_cfg = {
        .ctrl_if = ctrl_if,                              /* 控制接口(I2C) */
        .gpio_if = gpio_if,                              /* GPIO 接口 */
        .sys_cfg = {
            .is_master = false,                          /* codec 作 I2S 从机, 时钟由 ESP32 提供 */
            .no_mclk   = true,                           /* 不使用外部 MCLK */
        },
        .dac_cfg = {
            .ref_enable = true,                          /* 匹配 coeff 表 Ratio64 行 */
        },
        .pa_cfg = {
            .pa_pin        = -1,                         /* PA 由本驱动手动控制 */
            .pa_active_low = false,                      /* 高电平使能 PA */
            .hw_gain = {
                .pa_voltage        = 5.0,                /* PA 供电电压 5V */
                .codec_dac_voltage = 3.3,                /* codec DAC 供电电压 3.3V */
                .pa_gain           = 6.0,                /* PA 增益 6dB */
            },
        },
    };
    const audio_codec_if_t *es8389_if = es8389_codec_new(&es8389_cfg);
    assert(es8389_if);

    esp_codec_dev_cfg_t dev_cfg = {
        .dev_type = ESP_CODEC_DEV_TYPE_OUT,              /* 设备类型: 输出(播放) */
        .codec_if = es8389_if,                           /* codec 接口 */
        .data_if  = data_if,                             /* I2S 数据接口 */
    };
    s_codec_dev = esp_codec_dev_new(&dev_cfg);
    assert(s_codec_dev);

    if (es8389_dev_open(ES8389_SAMPLE_RATE, ES8389_BITS_PER_SAMPLE) != 0)
    {
        return 1;
    }

    if (esp_codec_dev_set_out_vol(s_codec_dev, ES8389_DEFAULT_VOLUME) != ESP_CODEC_DEV_OK)
    {
        ESP_LOGE(es8389_tag, "set output volume failed");
        return 1;
    }

    /* 开启功放 */
    gpio_config_t pa_cfg = {
        .pin_bit_mask = (1ULL << PA_CTRL_GPIO_PIN),      /* PA 使能引脚位掩码 */
        .mode         = GPIO_MODE_OUTPUT,                /* 输出模式 */
        .pull_up_en   = GPIO_PULLUP_DISABLE,             /* 不使用内部上拉 */
        .pull_down_en = GPIO_PULLDOWN_DISABLE,           /* 不使用内部下拉 */
        .intr_type    = GPIO_INTR_DISABLE,               /* 不使能中断 */
    };
    gpio_config(&pa_cfg);
    gpio_set_level(PA_CTRL_GPIO_PIN, 1);

    return 0;
}

/**
 * @brief       重新配置采样率(每首歌开始前 close+open 同步更新 I2S 时钟和 codec 系数)
 * @param       sample_rate      : 采样率(Hz)
 * @param       bits_per_sample  : 位宽(16/24)
 * @retval      0: 成功; 1: 不支持的采样率
 */
uint8_t myes8389_set_fs(uint32_t sample_rate, uint16_t bits_per_sample)
{
    esp_codec_dev_close(s_codec_dev);

    if (es8389_dev_open(sample_rate, bits_per_sample) != 0)
    {
        es8389_dev_open(ES8389_SAMPLE_RATE, ES8389_BITS_PER_SAMPLE);
        return 1;
    }

    return 0;
}

/**
 * @brief       设置输出音量
 * @param       volume : 0~100, 内部映射到 DAC 增益
 * @retval      无
 */
void myes8389_set_vol(int volume)
{
    esp_codec_dev_set_out_vol(s_codec_dev, volume);
}
