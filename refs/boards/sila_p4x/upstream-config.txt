# This project targets ESP32-P4 host only. IDF reads base defaults only from
# sdkconfig.defaults, so target defaults that must always apply live here.

CONFIG_ESP_MAIN_TASK_STACK_SIZE=8192
CONFIG_LOG_DEFAULT_LEVEL_INFO=y

# ESP32-P4X = ESP32-P4 chip revision v3.x (needs ESP-IDF >= v5.5.3 / v6.0).
# v3.x and v1.x firmware images are NOT interchangeable. REV_MIN_300 boots on
# both v3.0 and v3.1 (v3.0 needs a small MSPI workaround, auto-enabled by IDF).
# For old v1.x boards use: SELECTS_REV_LESS_V3=y, REV_MIN_100=y, CPU_FREQ_MHZ_360=y.
CONFIG_ESP32P4_SELECTS_REV_LESS_V3=n
CONFIG_ESP32P4_REV_MIN_300=y
CONFIG_ESP_DEFAULT_CPU_FREQ_MHZ_400=y

# ESP-Hosted: P4 is Host, C5 runs ESP-Hosted-MCU Slave 2.12.9.
CONFIG_SLAVE_IDF_TARGET_ESP32C5=y
CONFIG_ESP_HOSTED_IDF_SLAVE_TARGET="esp32c5"
CONFIG_ESP_HOSTED_SDIO_HOST_INTERFACE=y
CONFIG_ESP_HOSTED_SDIO_SLOT_1=y
# P4X board: 4-bit data reads are corrupted on D2/D3 (both in C5 ROM download mode and
# at runtime: "failed to read registers"), so Hosted runs on 1-bit (D0 data, D1 = IRQ).
CONFIG_ESP_HOSTED_SDIO_1_BIT_BUS=y
CONFIG_ESP_HOSTED_PRIV_SDIO_PIN_D1_1BIT_BUS_SLOT_1=15
# 诊断用：先降到 400kHz 验证是否为高速信号完整性问题；连通后再逐步升回（需配 10k 上拉）
CONFIG_ESP_HOSTED_SDIO_CLOCK_FREQ_KHZ=40000
CONFIG_ESP_HOSTED_PRIV_SDIO_PIN_CMD_SLOT_1=19
CONFIG_ESP_HOSTED_PRIV_SDIO_PIN_CLK_SLOT_1=18
CONFIG_ESP_HOSTED_PRIV_SDIO_PIN_D0_SLOT_1=14
# 4-bit pins kept for reference (D1=15 D2=16 D3=17); re-enable with ESP_HOSTED_SDIO_4_BIT_BUS=y
# once the D2/D3 signal issue is fixed in hardware.
CONFIG_ESP_HOSTED_SDIO_GPIO_RESET_SLAVE=54
# 对齐能跑通的官方范例：开机必复位 C5（让 slave 与 host 同步启动），
# 这是 ESP-Hosted 的默认且推荐策略，远比 RESET_ONLY_IF_NECESSARY 可靠。
CONFIG_ESP_HOSTED_SLAVE_RESET_ON_EVERY_HOST_BOOTUP=y
# 关闭传输失败自动重启系统：失败时只发事件、由 app 自行停下（默认是 y，必须显式关）
# CONFIG_ESP_HOSTED_TRANSPORT_RESTART_ON_FAILURE is not set
CONFIG_ESP_HOSTED_SDIO_RESET_DELAY_MS=1500

# P4-to-C5 first flash over SDIO.
# Old v1.x board: C5 GPIO25/MTDI strap floated, ROM SDIO download never answered
# (runtime SDIO 40 MHz was fine) -> UART via U14 was used instead ("plan A").
CONFIG_SERIAL_FLASHER_PORT_UART=n
CONFIG_SERIAL_FLASHER_PORT_SDIO=y
CONFIG_SERIAL_FLASHER_LOG_LEVEL_INFO=y
# Re-enabled for the ESP32-P4X board: P4 flashes a blank C5 over SDIO on first boot
# (10 MHz, 20 connect trials). Falls back to a warning if the C5 ROM does not answer.
CONFIG_P4_HOSTED_C5_FIRST_FLASH_ENABLE=y
CONFIG_P4_HOSTED_C5_FORCE_FLASH=n
CONFIG_P4_HOSTED_C5_RESET_GPIO=54
CONFIG_P4_HOSTED_C5_FLASH_FREQ_KHZ=10000
CONFIG_P4_HOSTED_C5_FLASH_CONNECT_TRIALS=20

# Flash and custom partition table (16MB flash for factory app + assets FAT)
CONFIG_ESPTOOLPY_FLASHSIZE_16MB=y
CONFIG_ESPTOOLPY_FLASHSIZE="16MB"
CONFIG_PARTITION_TABLE_CUSTOM=y
CONFIG_PARTITION_TABLE_CUSTOM_FILENAME="partitions.csv"

# AXS15260D 6.2in panel timing with 200MHz PSRAM and XIP
CONFIG_IDF_EXPERIMENTAL_FEATURES=y
CONFIG_ESP_TASK_WDT_EN=n
CONFIG_FREERTOS_HZ=1000
CONFIG_FREERTOS_TIMER_TASK_STACK_DEPTH=4096

# PSRAM for MIPI DSI frame buffers
CONFIG_SPIRAM=y
CONFIG_SPIRAM_MODE_HEX=y
CONFIG_SPIRAM_SPEED_200M=y
CONFIG_SPIRAM_XIP_FROM_PSRAM=y
CONFIG_CACHE_L2_CACHE_256KB=y
CONFIG_CACHE_L2_CACHE_LINE_128B=y
CONFIG_ESP_LDO_RESERVE_PSRAM=y
CONFIG_ESP_LDO_CHAN_PSRAM_DOMAIN=2
CONFIG_ESP_LDO_VOLTAGE_PSRAM_1800_MV=y
CONFIG_LCD_DSI_ISR_HANDLER_IN_IRAM=y

# FATFS Long File Name for assets partition
CONFIG_FATFS_LFN_HEAP=y

# USB Secondary Screen: ST7701 2.1in round (480x480, no touch)
# For AXS15260D 6.2in use APP_PANEL_AXS15260_6_2IN, touch on, 1280x452.
CONFIG_APP_USB_SCREEN_MODE=y
CONFIG_APP_PANEL_ST7701_2_1IN=y
CONFIG_USB_EXTEND_SCREEN_HEIGHT=480
CONFIG_USB_EXTEND_SCREEN_WIDTH=480
CONFIG_USB_EXTEND_SCREEN_JPEG_QUALITY=6
CONFIG_USB_EXTEND_SCREEN_MAX_FPS=40
CONFIG_USB_EXTEND_SCREEN_FRAME_LIMIT_B=2000000
CONFIG_USB_EXTEND_SCREEN_BUFFER_COUNT=4

# ES8389 Audio
CONFIG_CODEC_I2C_BACKWARD_COMPATIBLE=n
CONFIG_CODEC_ES8389_SUPPORT=y
CONFIG_P4_ES8389_ENABLE=y
CONFIG_P4_ES8389_I2S_BCLK_GPIO=12
CONFIG_P4_ES8389_I2S_WS_GPIO=10
CONFIG_P4_ES8389_I2S_DOUT_GPIO=9
CONFIG_P4_ES8389_PA_GPIO=31

# SC7A20 accelerometer test on I2C0 (SDA=IO7, SCL=IO8).

# AXP2101 interrupt test on I2C1 (SDA=IO28, SCL=IO29), AXP_IRQ=IO6. No rail is written.

CONFIG_P4_AXP2101_I2C_SDA_GPIO=28
CONFIG_P4_AXP2101_I2C_SCL_GPIO=29
CONFIG_P4_AXP2101_IRQ_GPIO=6

# Scan test parameters.
CONFIG_P4_HOSTED_C5_BOOT_RELEASE_GPIO=53
