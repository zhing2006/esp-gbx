    ESP_RETURN_ON_FALSE(s_bus == NULL, ESP_ERR_INVALID_STATE, TAG, "已初始化");
    s_irq_gpio = irq_gpio;

    // 1. I2C 总线 + 设备
    const i2c_master_bus_config_t bus_cfg = {
        .i2c_port = PMU_I2C_PORT,
        .sda_io_num = sda,
        .scl_io_num = scl,
        .clk_source = I2C_CLK_SRC_DEFAULT,
        .glitch_ignore_cnt = 7,
        .intr_priority = 0,
        .trans_queue_depth = 0,
        .flags = { .enable_internal_pullup = 1, .allow_pd = 0 },
    };
    ESP_RETURN_ON_ERROR(i2c_new_master_bus(&bus_cfg, &s_bus), TAG, "创建 I2C%d 总线失败", PMU_I2C_PORT);
    ESP_LOGI(TAG, "I2C%d 总线已创建：SDA=IO%d SCL=IO%d，用于 AXP2101", PMU_I2C_PORT, sda, scl);

    i2c_device_config_t dev_cfg = {};
    dev_cfg.dev_addr_length = I2C_ADDR_BIT_LEN_7;
    dev_cfg.device_address = PMU_I2C_ADDR;
    dev_cfg.scl_speed_hz = PMU_I2C_FREQ_HZ;
    ESP_RETURN_ON_ERROR(i2c_master_bus_add_device(s_bus, &dev_cfg, &s_dev), TAG, "添加 AXP2101 设备失败");
    vTaskDelay(pdMS_TO_TICKS(50));

    // 2. 识别芯片（不调用 XPowersLib::init()，它会顺带改 TS/充电等配置）
    s_pmu = new XPowersAXP2101(PMU_I2C_ADDR, pmu_component_read, pmu_component_write);
    ESP_RETURN_ON_FALSE(s_pmu, ESP_ERR_NO_MEM, TAG, "AXP2101 对象创建失败");
    const uint8_t chip_id = s_pmu->getChipID();
    if (chip_id != XPOWERS_AXP2101_CHIP_ID && chip_id != XPOWERS_AXP2101_CHIP_ID_2) {
        ESP_LOGE(TAG, "❌ AXP2101 芯片识别失败，读取ID=0x%02X（期望 0x%02X）", chip_id, XPOWERS_AXP2101_CHIP_ID);
        return ESP_FAIL;
    }
    ESP_LOGI(TAG, "🔧 AXP2101 芯片 ID: 0x%02X", chip_id);

    ESP_RETURN_ON_FALSE(s_pmu->disableIRQ(XPOWERS_AXP2101_ALL_IRQ), ESP_FAIL, TAG, "关闭旧 IRQ 使能失败");
    s_pmu->clearIrqStatus();
