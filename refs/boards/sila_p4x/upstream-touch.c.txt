static void touch_task(void *arg)
{
    (void)arg;
    int64_t read_error_since = 0;
    int64_t last_read_error_log = 0;

    while (1) {
        // 会话期（已按下或按下候选确认中）以固定周期跟踪；
        // 真空闲只等 INT，绝不盲读 I2C（避免指针漫游乱帧）。
        const bool session_active =
            s_filter.stable_pressed || s_filter.press_candidate.valid;
#if CONFIG_APP_PANEL_ST7701_2_8IN
        // 已验证的 CST826 分支采用轮询，避免依赖 GPIO27 中断极性。
        vTaskDelay(pdMS_TO_TICKS(TOUCH_PRESSED_POLL_MS));
#else
        const TickType_t wait = session_active
                                    ? pdMS_TO_TICKS(TOUCH_PRESSED_POLL_MS)
                                    : portMAX_DELAY;
        const uint32_t notified = ulTaskNotifyTake(pdTRUE, wait);
        if (!notified && !session_active) {
            continue;
        }
#endif

        touch_sample_t raw = {0};
        bool raw_valid = false;
#if CONFIG_APP_PANEL_ST7701_2_8IN
        uint16_t px = 0, py = 0;
        bool pressed_now = false;
        esp_err_t read_ret = cst826_touch_read(&px, &py, &pressed_now);
        if (read_ret == ESP_OK && pressed_now) {
            raw = (touch_sample_t){ .x = px, .y = py, .size = TOUCH_DEFAULT_SIZE,
                                    .id = 0, .valid = true };
            raw_valid = true;
        }
#else
        axs15260_touch_data_t data = {0};
        esp_err_t read_ret = axs15260_touch_read(s_tp, &data);
        if (read_ret == ESP_OK && data.point_num > 0) {
            const axs15260_touch_point_t *pt = &data.points[0];
            if (pt->x < TOUCH_PANEL_H_RES && pt->y < TOUCH_PANEL_V_RES) {
                raw.x = pt->x;
                raw.y = pt->y;
                raw.size = pt->area ? pt->area : pt->weight;
                raw.id = pt->id;
                raw.valid = true;
                raw_valid = true;
            }
        }
#endif

#include "cst826_touch.h"
#include "esp_check.h"
#include "esp_log.h"

static const char *TAG = "cst826_touch";
static i2c_master_dev_handle_t s_device;

esp_err_t cst826_touch_init(i2c_master_bus_handle_t bus)
{
    ESP_RETURN_ON_FALSE(bus && !s_device, ESP_ERR_INVALID_STATE, TAG, "触摸总线为空或重复初始化");
    // GPIO26 与 LCD 共用复位线，触摸初始化不再操作该引脚。
    i2c_device_config_t cfg = {
        .dev_addr_length = I2C_ADDR_BIT_LEN_7,
        .device_address = 0x15,
        .scl_speed_hz = 400000,
    };
    ESP_RETURN_ON_ERROR(i2c_master_bus_add_device(bus, &cfg, &s_device), TAG, "添加 CST826 设备失败");
    uint8_t reg = 0xA7, id = 0;
    esp_err_t ret = i2c_master_transmit_receive(s_device, &reg, 1, &id, 1, 100);
    if (ret != ESP_OK) {
        i2c_master_bus_rm_device(s_device);
        s_device = NULL;
        ESP_LOGE(TAG, "读取 CST826 芯片 ID 失败：%s", esp_err_to_name(ret));
        return ret;
    }
    // 分支记录兼容批次 ID 不同，因此保留后续触摸事件验证。
    if (id != 0xCA) ESP_LOGW(TAG, "CST826 芯片 ID 与参考值不同：实际=0x%02X，参考=0xCA", id);
    ESP_LOGI(TAG, "CST826 触摸通信已就绪：地址=0x15，ID=0x%02X，采用 20ms 轮询", id);
    return ESP_OK;
}

esp_err_t cst826_touch_read(uint16_t *x, uint16_t *y, bool *pressed)
{
    if (!s_device || !x || !y || !pressed) return ESP_ERR_INVALID_ARG;
    *pressed = false;
    uint8_t reg = 0x02, data[13] = {0};
