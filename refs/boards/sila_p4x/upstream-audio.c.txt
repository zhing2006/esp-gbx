#define I2S_GPIO_BCLK         ((gpio_num_t)CONFIG_P4_ES8389_I2S_BCLK_GPIO)
#define I2S_GPIO_WS           ((gpio_num_t)CONFIG_P4_ES8389_I2S_WS_GPIO)
#define I2S_GPIO_DOUT         ((gpio_num_t)CONFIG_P4_ES8389_I2S_DOUT_GPIO)
#define CODEC_PA_PIN          ((gpio_num_t)CONFIG_P4_ES8389_PA_GPIO)

static i2s_chan_handle_t s_tx_chan = NULL;

static esp_err_t pa_ctrl_init(void)
{
    if (CODEC_PA_PIN == GPIO_NUM_NC) {
        return ESP_OK;
    }
    const gpio_config_t cfg = {
        .pin_bit_mask = 1ULL << CODEC_PA_PIN,
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE,
    };
    ESP_RETURN_ON_ERROR(gpio_config(&cfg), TAG, "❌ 配置 PA 使能引脚失败");
    return gpio_set_level(CODEC_PA_PIN, 0);
}

static void pa_ctrl_enable(bool on)
{
    if (CODEC_PA_PIN != GPIO_NUM_NC) {
        gpio_set_level(CODEC_PA_PIN, on ? 1 : 0);
    }
}

static esp_err_t i2s_tx_init(void)
{
    if (s_tx_chan != NULL) {
        return ESP_OK;
    }

    i2s_chan_config_t chan_cfg = I2S_CHANNEL_DEFAULT_CONFIG(AUDIO_I2S_PORT, I2S_ROLE_MASTER);
    chan_cfg.dma_desc_num = 4;
    chan_cfg.dma_frame_num = 240;
    chan_cfg.auto_clear = true;
    ESP_RETURN_ON_ERROR(i2s_new_channel(&chan_cfg, &s_tx_chan, NULL), TAG, "❌ 创建 I2S TX 通道失败");

    const i2s_std_config_t std_cfg = {
        .clk_cfg = {
            .sample_rate_hz = AUDIO_SAMPLE_RATE,
            .clk_src = I2S_CLK_SRC_DEFAULT,
            .mclk_multiple = I2S_MCLK_MULTIPLE_256,
        },
        .slot_cfg = I2S_STD_PHILIPS_SLOT_DEFAULT_CONFIG(I2S_DATA_BIT_WIDTH_16BIT,
                                                        I2S_SLOT_MODE_STEREO),
        .gpio_cfg = {
            .mclk = I2S_GPIO_UNUSED,        // 本板 MCLK 未连线，Codec 从 BCLK/WS 派生
            .bclk = I2S_GPIO_BCLK,
            .ws   = I2S_GPIO_WS,
            .dout = I2S_GPIO_DOUT,
            .din  = I2S_GPIO_UNUSED,
            .invert_flags = { false, false, false },
        },
    };
    esp_err_t ret = i2s_channel_init_std_mode(s_tx_chan, &std_cfg);
    // esp_codec_dev 1.5.11 的数据接口接管时会先禁用通道；沿用组件示例先启用的时序。
    if (ret == ESP_OK) ret = i2s_channel_enable(s_tx_chan);
